1.Create a function to calculate price_range(low, medium, high) of each product. If price<=5
mysql> create function price(p decimal(5,2))
    -> returns varchar(10)
    -> deterministic
    -> begin
    -> declare result varchar(10);
    -> if p<=5 then
    -> set result='low';
    -> elseif p<=50 then
    -> set result='medium';
    -> else
    -> set result='high';
    -> end if;
    -> return result;
    -> end //
mysql> select pname,price,price_r(price) as price_category from product;//
