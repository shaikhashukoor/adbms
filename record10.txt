1. Create a procedure to find greatest of 3 numbers. 

mysql> create procedure bigest(a int,b int ,c int)
    -> begin
    -> declare x int;
    -> set x=a;
    -> if x<b then
    -> set x=b;
    -> elseif x<c then
    -> set x=c;
    -> end if;
    -> select x as greatest_no;
    -> end //
